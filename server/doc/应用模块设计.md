# 应用模块设计

## 功能概述

应用模块是系统的核心模块，负责管理系统中的应用插件。主要功能包括：

1. 应用的安装、卸载、启用、禁用
2. 应用版本管理
3. 应用依赖管理
4. 应用资源管理
5. 应用配置管理

## 实现细节

### 应用安装流程

1. 上传应用插件包（JAR文件）
2. 解析插件包，提取元数据
3. 检查应用是否已存在
   - 如果存在，检查是否为更新操作
   - 如果是更新，检查版本是否兼容
4. 检查依赖是否满足
5. 创建应用记录
6. 注册应用提供的权限、菜单、API和资源
7. 创建应用所需的数据库表
8. 初始化应用配置
9. 调用插件的onInstall方法
10. 启用应用

```kotlin
suspend fun installApplication(pluginFile: File, installBy: String): Application {
    // 解析插件包
    val pluginJar = JarFile(pluginFile)
    val metadata = pluginLoader.loadMetadata(pluginJar)
    
    // 检查应用是否已存在
    val existingApp = applicationRepository.findByCode(metadata.code)
    if (existingApp != null) {
        if (existingApp.status == ApplicationStatus.UNINSTALLED) {
            // 重新安装已卸载的应用
            return reinstallApplication(existingApp.id, pluginFile, installBy)
        } else {
            // 更新已安装的应用
            return updateApplication(existingApp.id, pluginFile, installBy)
        }
    }
    
    // 检查依赖
    val dependencies = metadata.dependencies
    for (dependency in dependencies) {
        val depApp = applicationRepository.findByCode(dependency.pluginId)
        if (depApp == null) {
            if (!dependency.isOptional) {
                throw DependencyNotFoundException("依赖应用 ${dependency.pluginId} 未安装")
            }
        } else {
            if (!versionManager.isCompatible(depApp.version, dependency.version)) {
                throw IncompatibleDependencyException("依赖应用 ${dependency.pluginId} 版本不兼容")
            }
        }
    }
    
    // 创建应用记录
    val application = Application(
        name = metadata.name,
        code = metadata.code,
        shortCode = metadata.shortCode,
        version = metadata.version,
        description = metadata.description,
        author = metadata.author,
        status = ApplicationStatus.INSTALLED,
        installTime = LocalDateTime.now(),
        createdBy = installBy,
        updatedBy = installBy
    )
    val savedApp = applicationRepository.save(application)
    
    // 加载插件
    val plugin = pluginLoader.loadPlugin(pluginJar)
    
    // 注册依赖
    for (dependency in metadata.dependencies) {
        val depApp = applicationRepository.findByCode(dependency.pluginId)
        if (depApp != null) {
            val appDependency = ApplicationDependency(
                applicationId = savedApp.id,
                dependencyId = depApp.id,
                version = dependency.version,
                isOptional = dependency.isOptional,
                createdBy = installBy,
                updatedBy = installBy
            )
            applicationDependencyRepository.save(appDependency)
        }
    }
    
    // 注册权限、菜单、API和资源
    registerPluginComponents(plugin, savedApp.id, installBy)
    
    // 创建数据库表
    createPluginTables(plugin, metadata.shortCode)
    
    // 初始化配置
    initializePluginConfigs(plugin, savedApp.id, installBy)
    
    // 调用插件安装方法
    plugin.onInstall()
    
    // 启用插件
    plugin.onEnable()
    
    return savedApp
}

private suspend fun registerPluginComponents(plugin: KAcePlugin, applicationId: String, createdBy: String) {
    // 注册权限
    val permissions = plugin.registerPermissions()
    for (permission in permissions) {
        permissionService.createPermission(Permission(
            name = permission.name,
            code = permission.code,
            type = PermissionType.valueOf(permission.type.name),
            applicationId = applicationId,
            description = permission.description,
            createdBy = createdBy,
            updatedBy = createdBy
        ))
    }
    
    // 注册菜单
    val menus = plugin.registerMenus()
    for (menu in menus) {
        val permission = permissionService.getPermissionByCode(menu.permissionCode)
        menuService.createMenu(Menu(
            name = menu.name,
            code = menu.code,
            path = menu.path,
            icon = menu.icon,
            parentId = menuService.getMenuByCode(menu.parentCode)?.id,
            applicationId = applicationId,
            sort = menu.sort,
            permissionId = permission?.id,
            createdBy = createdBy,
            updatedBy = createdBy
        ))
    }
    
    // 注册API
    val apis = plugin.registerApis()
    for (api in apis) {
        val permission = permissionService.getPermissionByCode(api.permissionCode)
        apiService.createApi(Api(
            path = api.path,
            method = HttpMethod.valueOf(api.method.name),
            permissionId = permission?.id,
            applicationId = applicationId,
            description = api.description,
            createdBy = createdBy,
            updatedBy = createdBy
        ))
    }
    
    // 注册资源
    val resources = plugin.registerResources()
    for (resource in resources) {
        val content = plugin.getResourceAsStream(resource.path)
        applicationResourceService.addResource(ApplicationResource(
            applicationId = applicationId,
            version = plugin.metadata.version,
            type = ResourceType.valueOf(resource.type.name),
            path = resource.path,
            name = resource.path.substringAfterLast('/'),
            size = content.available().toLong(),
            hash = DigestUtils.sha256Hex(content),
            createdBy = createdBy,
            updatedBy = createdBy
        ), content)
    }
}

private suspend fun createPluginTables(plugin: KAcePlugin, shortCode: String) {
    val tables = plugin.registerTables()
    for (table in tables) {
        val tableName = "${shortCode}_${table.name}"
        val columns = table.columns.map { column ->
            "`${column.name}` ${column.type} ${if (!column.nullable) "NOT NULL" else ""} ${column.defaultValue?.let { "DEFAULT $it" } ?: ""}"
        }
        val primaryKey = "PRIMARY KEY (`${table.primaryKey}`)"
        
        val createTableSql = """
            CREATE TABLE `$tableName` (
                ${columns.joinToString(",\n                ")},
                $primaryKey
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
        """.trimIndent()
        
        databaseService.executeUpdate(createTableSql)
        
        // 创建索引
        for (index in table.indices) {
            val indexSql = """
                CREATE ${if (index.unique) "UNIQUE " else ""}INDEX `${index.name}` ON `$tableName` (${index.columns.joinToString(", ") { "`$it`" }})
            """.trimIndent()
            databaseService.executeUpdate(indexSql)
        }
    }
}

private suspend fun initializePluginConfigs(plugin: KAcePlugin, applicationId: String, createdBy: String) {
    val configs = plugin.registerConfigs()
    for (config in configs) {
        applicationConfigService.setConfig(ApplicationConfig(
            applicationId = applicationId,
            key = config.key,
            value = config.defaultValue,
            type = ConfigType.valueOf(config.type.name),
            description = config.description,
            isSystem = config.isSystem,
            createdBy = createdBy,
            updatedBy = createdBy
        ))
    }
}
```

### 应用卸载流程

1. 检查是否有其他应用依赖该应用
2. 调用插件的onDisable方法
3. 调用插件的onUninstall方法
4. 更新应用状态为已卸载

```kotlin
suspend fun uninstallApplication(id: String, uninstallBy: String): Boolean {
    val application = applicationRepository.findById(id) ?: throw ApplicationNotFoundException("应用不存在")
    
    // 检查依赖
    val dependentApps = applicationDependencyRepository.findByDependencyId(id)
    if (dependentApps.isNotEmpty()) {
        val nonOptionalDeps = dependentApps.filter { !it.isOptional }
        if (nonOptionalDeps.isNotEmpty()) {
            val appNames = nonOptionalDeps.map { 
                applicationRepository.findById(it.applicationId)?.name ?: it.applicationId
            }
            throw DependencyViolationException("以下应用依赖此应用，无法卸载：${appNames.joinToString(", ")}")
        }
    }
    
    // 加载插件
    val plugin = pluginManager.getPlugin(application.code) ?: throw PluginNotFoundException("插件不存在")
    
    // 禁用插件
    plugin.onDisable()
    
    // 卸载插件
    plugin.onUninstall()
    
    // 更新应用状态
    application.status = ApplicationStatus.UNINSTALLED
    application.uninstallTime = LocalDateTime.now()
    application.updatedBy = uninstallBy
    applicationRepository.update(application)
    
    return true
}
```

### 资源存储

系统支持多种存储方式，默认使用本地文件系统：

```kotlin
interface StorageProvider {
    fun store(path: String, content: InputStream, contentType: String): String
    fun retrieve(path: String): InputStream
    fun delete(path: String): Boolean
    fun exists(path: String): Boolean
}

class LocalStorageProvider(private val baseDir: String) : StorageProvider {
    override fun store(path: String, content: InputStream, contentType: String): String {
        val file = File("$baseDir/$path")
        file.parentFile.mkdirs()
        FileOutputStream(file).use { output ->
            content.copyTo(output)
        }
        return path
    }
    
    override fun retrieve(path: String): InputStream {
        val file = File("$baseDir/$path")
        if (!file.exists()) {
            throw FileNotFoundException("文件不存在：$path")
        }
        return FileInputStream(file)
    }
    
    override fun delete(path: String): Boolean {
        val file = File("$baseDir/$path")
        return file.delete()
    }
    
    override fun exists(path: String): Boolean {
        val file = File("$baseDir/$path")
        return file.exists()
    }
}

class TencentCosStorageProvider(
    private val secretId: String,
    private val secretKey: String,
    private val region: String,
    private val bucketName: String
) : StorageProvider {
    private val cosClient = COSClient(
        BasicCOSCredentials(secretId, secretKey),
        ClientConfig(Region(region))
    )
    
    override fun store(path: String, content: InputStream, contentType: String): String {
        val metadata = ObjectMetadata()
        metadata.contentType = contentType
        
        val putObjectRequest = PutObjectRequest(bucketName, path, content, metadata)
        cosClient.putObject(putObjectRequest)
        
        return path
    }
    
    override fun retrieve(path: String): InputStream {
        val getObjectRequest = GetObjectRequest(bucketName, path)
        val cosObject = cosClient.getObject(getObjectRequest)
        return cosObject.objectContent
    }
    
    override fun delete(path: String): Boolean {
        cosClient.deleteObject(bucketName, path)
        return true
    }
    
    override fun exists(path: String): Boolean {
        return try {
            cosClient.getObjectMetadata(bucketName, path)
            true
        } catch (e: CosServiceException) {
            if (e.statusCode == 404) {
                false
            } else {
                throw e
            }
        }
    }
}

class AliyunOssStorageProvider(
    private val endpoint: String,
    private val accessKeyId: String,
    private val accessKeySecret: String,
    private val bucketName: String
) : StorageProvider {
    private val ossClient = OSSClientBuilder().build(endpoint, accessKeyId, accessKeySecret)
    
    override fun store(path: String, content: InputStream, contentType: String): String {
        val metadata = ObjectMetadata()
        metadata.contentType = contentType
        
        ossClient.putObject(bucketName, path, content, metadata)
        
        return path
    }
    
    override fun retrieve(path: String): InputStream {
        val ossObject = ossClient.getObject(bucketName, path)
        return ossObject.objectContent
    }
    
    override fun delete(path: String): Boolean {
        ossClient.deleteObject(bucketName, path)
        return true
    }
    
    override fun exists(path: String): Boolean {
        return ossClient.doesObjectExist(bucketName, path)
    }
}
```

## 安全考虑

1. **应用隔离**：确保应用只能访问自己的资源和配置
2. **依赖管理**：严格检查应用依赖，防止依赖冲突
3. **版本控制**：维护应用版本历史，支持版本回滚
4. **资源验证**：验证应用资源的完整性，防止资源被篡改
5. **配置保护**：系统配置不允许通过UI修改，防止误操作
