# 数据库设计

## 基础实体模型

所有实体都应包含以下基础字段：
- id: VARCHAR(36) (主键，使用UUID格式)
- createdAt: DATETIME (创建时间)
- updatedAt: DATETIME (修改时间)
- deletedAt: DATETIME (删除时间，软删除标记，NULL表示未删除)
- createdBy: VARCHAR(36) (创建人ID)
- updatedBy: VARCHAR(36) (修改人ID)
- deletedBy: VARCHAR(36) (删除人ID)

## 用户模块

### 用户表 (sys_user)
- username: VARCHAR(50) (用户名，唯一)
- password: VARCHAR(100) (密码，加密存储)
- status: VARCHAR(10) (状态：ENABLED/DISABLED)
- lastLoginAt: DATETIME (最后登录时间)
- lastLoginIp: VARCHAR(50) (最后登录IP)

### 用户角色关联表 (sys_user_role)
- userId: VARCHAR(36) (用户ID)
- roleId: VARCHAR(36) (角色ID)

## 角色模块

### 角色表 (sys_role)
- name: VARCHAR(50) (角色名称)
- code: VARCHAR(50) (角色编码，唯一)
- description: VARCHAR(200) (角色描述)
- status: VARCHAR(10) (状态：ENABLED/DISABLED)

### 角色应用关联表 (sys_role_application)
- roleId: VARCHAR(36) (角色ID)
- applicationId: VARCHAR(36) (应用ID)

### 角色权限表 (sys_role_permission)
- roleId: VARCHAR(36) (角色ID)
- permissionId: VARCHAR(36) (权限ID)

## 权限模块

### 权限表 (sys_permission)
- name: VARCHAR(100) (权限名称)
- code: VARCHAR(100) (权限编码，唯一)
- type: VARCHAR(20) (权限类型：API/RESOURCE)
- applicationId: VARCHAR(36) (所属应用ID)
- description: VARCHAR(200) (权限描述)

### 菜单表 (sys_menu)
- name: VARCHAR(100) (菜单名称)
- code: VARCHAR(100) (菜单编码，唯一)
- path: VARCHAR(200) (菜单路径)
- icon: VARCHAR(100) (菜单图标)
- parentId: VARCHAR(36) (父菜单ID)
- applicationId: VARCHAR(36) (所属应用ID)
- sort: INT (排序值)
- permissionId: VARCHAR(36) (关联的权限ID)

### API表 (sys_api)
- path: VARCHAR(200) (API路径)
- method: VARCHAR(10) (HTTP方法)
- permissionId: VARCHAR(36) (关联的权限ID)
- applicationId: VARCHAR(36) (所属应用ID)
- description: VARCHAR(200) (API描述)

## 应用模块

### 应用表 (sys_application)
- name: VARCHAR(100) (应用名称)
- code: VARCHAR(100) (应用编码，唯一)
- shortCode: VARCHAR(20) (应用简码，用于表前缀)
- version: VARCHAR(20) (应用版本)
- description: VARCHAR(500) (应用描述)
- author: VARCHAR(100) (应用作者)
- homepage: VARCHAR(200) (应用主页)
- icon: VARCHAR(200) (应用图标)
- status: VARCHAR(20) (状态：INSTALLED/UNINSTALLED/DISABLED)
- installTime: DATETIME (安装时间)
- uninstallTime: DATETIME (卸载时间)

### 应用版本表 (sys_application_version)
- applicationId: VARCHAR(36) (应用ID)
- version: VARCHAR(20) (版本号)
- description: VARCHAR(500) (版本描述)
- releaseTime: DATETIME (发布时间)
- status: VARCHAR(20) (状态：RELEASED/DEPRECATED)

### 应用依赖表 (sys_application_dependency)
- applicationId: VARCHAR(36) (应用ID)
- dependencyId: VARCHAR(36) (依赖应用ID)
- version: VARCHAR(20) (依赖版本要求)
- isOptional: TINYINT(1) (是否可选依赖，1表示是，0表示否)

### 应用资源表 (sys_application_resource)
- applicationId: VARCHAR(36) (应用ID)
- version: VARCHAR(20) (应用版本)
- type: VARCHAR(20) (资源类型：IMAGE/FILE/OTHER)
- path: VARCHAR(500) (资源路径)
- name: VARCHAR(200) (资源名称)
- size: BIGINT (资源大小)
- hash: VARCHAR(64) (资源哈希值)

### 应用配置表 (sys_application_config)
- applicationId: VARCHAR(36) (应用ID)
- `key`: VARCHAR(100) (配置键)
- `value`: TEXT (配置值)
- type: VARCHAR(20) (配置类型：STRING/NUMBER/BOOLEAN/JSON)
- description: VARCHAR(200) (配置描述)
- isSystem: TINYINT(1) (是否系统配置，1表示是，0表示否)

## 日志模块

### 操作日志表 (sys_operation_log)
- userId: VARCHAR(36) (操作用户ID)
- username: VARCHAR(50) (操作用户名)
- applicationId: VARCHAR(36) (应用ID)
- module: VARCHAR(50) (操作模块)
- operation: VARCHAR(50) (操作类型)
- method: VARCHAR(10) (请求方法)
- path: VARCHAR(200) (请求路径)
- params: TEXT (请求参数)
- result: TEXT (操作结果)
- status: INT (状态码)
- errorMsg: TEXT (错误信息)
- ip: VARCHAR(50) (操作IP)
- userAgent: VARCHAR(200) (用户代理)
- duration: BIGINT (操作耗时，单位ms)
- createdAt: DATETIME (创建时间)

### 日志配置表 (sys_log_config)
- applicationId: VARCHAR(36) (应用ID)
- level: VARCHAR(10) (日志级别：TRACE/DEBUG/INFO/WARN/ERROR)
- path: VARCHAR(200) (日志路径)
- maxFileSize: BIGINT (单个日志文件最大大小，默认10MB)
- maxHistory: INT (日志保留天数，默认30天)
- isAsync: TINYINT(1) (是否异步记录，1表示是，0表示否)

## SQL建表语句

```sql
-- 用户表
CREATE TABLE `sys_user` (
    `id` VARCHAR(36) PRIMARY KEY,
    `username` VARCHAR(50) NOT NULL UNIQUE,
    `password` VARCHAR(100) NOT NULL,
    `status` VARCHAR(10) NOT NULL DEFAULT 'ENABLED',
    `last_login_at` DATETIME,
    `last_login_ip` VARCHAR(50),
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 用户角色关联表
CREATE TABLE `sys_user_role` (
    `id` VARCHAR(36) PRIMARY KEY,
    `user_id` VARCHAR(36) NOT NULL,
    `role_id` VARCHAR(36) NOT NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_user_role` (`user_id`, `role_id`),
    CONSTRAINT `fk_user_role_user` FOREIGN KEY (`user_id`) REFERENCES `sys_user` (`id`),
    CONSTRAINT `fk_user_role_role` FOREIGN KEY (`role_id`) REFERENCES `sys_role` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 角色表
CREATE TABLE `sys_role` (
    `id` VARCHAR(36) PRIMARY KEY,
    `name` VARCHAR(50) NOT NULL,
    `code` VARCHAR(50) NOT NULL UNIQUE,
    `description` VARCHAR(200),
    `status` VARCHAR(10) NOT NULL DEFAULT 'ENABLED',
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 角色应用关联表
CREATE TABLE `sys_role_application` (
    `id` VARCHAR(36) PRIMARY KEY,
    `role_id` VARCHAR(36) NOT NULL,
    `application_id` VARCHAR(36) NOT NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_role_app` (`role_id`, `application_id`),
    CONSTRAINT `fk_role_app_role` FOREIGN KEY (`role_id`) REFERENCES `sys_role` (`id`),
    CONSTRAINT `fk_role_app_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 角色权限表
CREATE TABLE `sys_role_permission` (
    `id` VARCHAR(36) PRIMARY KEY,
    `role_id` VARCHAR(36) NOT NULL,
    `permission_id` VARCHAR(36) NOT NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_role_perm` (`role_id`, `permission_id`),
    CONSTRAINT `fk_role_perm_role` FOREIGN KEY (`role_id`) REFERENCES `sys_role` (`id`),
    CONSTRAINT `fk_role_perm_perm` FOREIGN KEY (`permission_id`) REFERENCES `sys_permission` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 权限表
CREATE TABLE `sys_permission` (
    `id` VARCHAR(36) PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `code` VARCHAR(100) NOT NULL UNIQUE,
    `type` VARCHAR(20) NOT NULL,
    `application_id` VARCHAR(36) NOT NULL,
    `description` VARCHAR(200),
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    CONSTRAINT `fk_perm_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 菜单表
CREATE TABLE `sys_menu` (
    `id` VARCHAR(36) PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `code` VARCHAR(100) NOT NULL UNIQUE,
    `path` VARCHAR(200),
    `icon` VARCHAR(100),
    `parent_id` VARCHAR(36),
    `application_id` VARCHAR(36) NOT NULL,
    `sort` INT NOT NULL DEFAULT 0,
    `permission_id` VARCHAR(36),
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    CONSTRAINT `fk_menu_parent` FOREIGN KEY (`parent_id`) REFERENCES `sys_menu` (`id`),
    CONSTRAINT `fk_menu_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`),
    CONSTRAINT `fk_menu_perm` FOREIGN KEY (`permission_id`) REFERENCES `sys_permission` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- API表
CREATE TABLE `sys_api` (
    `id` VARCHAR(36) PRIMARY KEY,
    `path` VARCHAR(200) NOT NULL,
    `method` VARCHAR(10) NOT NULL,
    `permission_id` VARCHAR(36),
    `application_id` VARCHAR(36) NOT NULL,
    `description` VARCHAR(200),
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_path_method_app` (`path`, `method`, `application_id`),
    CONSTRAINT `fk_api_perm` FOREIGN KEY (`permission_id`) REFERENCES `sys_permission` (`id`),
    CONSTRAINT `fk_api_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 应用表
CREATE TABLE `sys_application` (
    `id` VARCHAR(36) PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `code` VARCHAR(100) NOT NULL UNIQUE,
    `short_code` VARCHAR(20) NOT NULL UNIQUE,
    `version` VARCHAR(20) NOT NULL,
    `description` VARCHAR(500),
    `author` VARCHAR(100),
    `homepage` VARCHAR(200),
    `icon` VARCHAR(200),
    `status` VARCHAR(20) NOT NULL DEFAULT 'INSTALLED',
    `install_time` DATETIME,
    `uninstall_time` DATETIME,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 应用版本表
CREATE TABLE `sys_application_version` (
    `id` VARCHAR(36) PRIMARY KEY,
    `application_id` VARCHAR(36) NOT NULL,
    `version` VARCHAR(20) NOT NULL,
    `description` VARCHAR(500),
    `release_time` DATETIME NOT NULL,
    `status` VARCHAR(20) NOT NULL DEFAULT 'RELEASED',
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_app_version` (`application_id`, `version`),
    CONSTRAINT `fk_app_version_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 应用依赖表
CREATE TABLE `sys_application_dependency` (
    `id` VARCHAR(36) PRIMARY KEY,
    `application_id` VARCHAR(36) NOT NULL,
    `dependency_id` VARCHAR(36) NOT NULL,
    `version` VARCHAR(20) NOT NULL,
    `is_optional` TINYINT(1) NOT NULL DEFAULT 0,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_app_dependency` (`application_id`, `dependency_id`),
    CONSTRAINT `fk_app_dep_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`),
    CONSTRAINT `fk_app_dep_dep` FOREIGN KEY (`dependency_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 应用资源表
CREATE TABLE `sys_application_resource` (
    `id` VARCHAR(36) PRIMARY KEY,
    `application_id` VARCHAR(36) NOT NULL,
    `version` VARCHAR(20) NOT NULL,
    `type` VARCHAR(20) NOT NULL,
    `path` VARCHAR(500) NOT NULL,
    `name` VARCHAR(200) NOT NULL,
    `size` BIGINT NOT NULL,
    `hash` VARCHAR(64) NOT NULL,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    CONSTRAINT `fk_app_res_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 应用配置表
CREATE TABLE `sys_application_config` (
    `id` VARCHAR(36) PRIMARY KEY,
    `application_id` VARCHAR(36) NOT NULL,
    `key` VARCHAR(100) NOT NULL,
    `value` TEXT NOT NULL,
    `type` VARCHAR(20) NOT NULL,
    `description` VARCHAR(200),
    `is_system` TINYINT(1) NOT NULL DEFAULT 0,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_app_key` (`application_id`, `key`),
    CONSTRAINT `fk_app_config_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 操作日志表
CREATE TABLE `sys_operation_log` (
    `id` VARCHAR(36) PRIMARY KEY,
    `user_id` VARCHAR(36),
    `username` VARCHAR(50),
    `application_id` VARCHAR(36),
    `module` VARCHAR(50),
    `operation` VARCHAR(50),
    `method` VARCHAR(10),
    `path` VARCHAR(200),
    `params` TEXT,
    `result` TEXT,
    `status` INT,
    `error_msg` TEXT,
    `ip` VARCHAR(50),
    `user_agent` VARCHAR(200),
    `duration` BIGINT,
    `created_at` DATETIME NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 日志配置表
CREATE TABLE `sys_log_config` (
    `id` VARCHAR(36) PRIMARY KEY,
    `application_id` VARCHAR(36) NOT NULL,
    `level` VARCHAR(10) NOT NULL DEFAULT 'INFO',
    `path` VARCHAR(200),
    `max_file_size` BIGINT NOT NULL DEFAULT 10485760, -- 10MB
    `max_history` INT NOT NULL DEFAULT 30,
    `is_async` TINYINT(1) NOT NULL DEFAULT 1,
    `created_at` DATETIME NOT NULL,
    `updated_at` DATETIME NOT NULL,
    `deleted_at` DATETIME,
    `created_by` VARCHAR(36),
    `updated_by` VARCHAR(36),
    `deleted_by` VARCHAR(36),
    UNIQUE KEY `uk_app_log` (`application_id`),
    CONSTRAINT `fk_log_config_app` FOREIGN KEY (`application_id`) REFERENCES `sys_application` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
