# 日志模块设计

## 功能概述

日志模块负责系统和应用的日志记录、存储和查询。主要功能包括：

1. 操作日志记录
2. 应用日志管理
3. 日志配置管理
4. 日志查询和分析

## 数据模型

## 实现细节

### 操作日志记录

系统使用AOP（面向切面编程）自动记录操作日志：

```kotlin
@Aspect
class OperationLogAspect(
    private val operationLogService: OperationLogService,
    private val userContextHolder: UserContextHolder
) {
    @Around("@annotation(operationLog)")
    suspend fun around(joinPoint: ProceedingJoinPoint, operationLog: OperationLog): Any? {
        val startTime = System.currentTimeMillis()
        val request = getCurrentRequest()
        val userContext = userContextHolder.getContext()
        
        var result: Any? = null
        var error: Throwable? = null
        
        try {
            result = joinPoint.proceed()
            return result
        } catch (e: Throwable) {
            error = e
            throw e
        } finally {
            val duration = System.currentTimeMillis() - startTime
            
            val log = OperationLog(
                userId = userContext?.userId,
                username = userContext?.username,
                applicationId = operationLog.applicationId,
                module = operationLog.module,
                operation = operationLog.operation,
                method = request.method.value,
                path = request.path(),
                params = request.queryParameters.toMap().toString(),
                result = result?.toString(),
                status = if (error == null) 200 else 500,
                errorMsg = error?.message,
                ip = request.origin.remoteHost,
                userAgent = request.headers["User-Agent"],
                duration = duration
            )
            
            // 异步记录日志
            launch {
                operationLogService.createLog(log)
            }
        }
    }
}
```

### 应用日志配置

每个应用有独立的日志配置和日志文件：

```kotlin
class LogbackConfigurer(
    private val logConfigService: LogConfigService
) {
    suspend fun configureLogback() {
        val logConfigs = logConfigService.listConfigs(1, Int.MAX_VALUE).content
        
        val loggerContext = LoggerFactory.getILoggerFactory() as LoggerContext
        
        for (config in logConfigs) {
            val applicationId = config.applicationId.toString()
            val logger = loggerContext.getLogger("application.$applicationId")
            
            // 设置日志级别
            logger.level = Level.toLevel(config.level.name)
            
            // 创建文件appender
            val fileAppender = FileAppender<ILoggingEvent>().apply {
                context = loggerContext
                name = "FILE-$applicationId"
                file = config.path ?: "logs/application-$applicationId.log"
                
                // 设置日志格式
                encoder = PatternLayoutEncoder().apply {
                    context = loggerContext
                    pattern = "%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n"
                    start()
                }
                
                // 设置滚动策略
                rollingPolicy = TimeBasedRollingPolicy<ILoggingEvent>().apply {
                    context = loggerContext
                    fileNamePattern = "${file}.%d{yyyy-MM-dd}.%i.gz"
                    maxHistory = config.maxHistory
                    timeBasedFileNamingAndTriggeringPolicy = SizeAndTimeBasedFNATP<ILoggingEvent>().apply {
                        context = loggerContext
                        maxFileSize = FileSize.valueOf("${config.maxFileSize}b")
                    }
                    setParent(this@apply)
                    start()
                }
                
                // 设置异步
                if (config.isAsync) {
                    val asyncAppender = AsyncAppender().apply {
                        context = loggerContext
                        name = "ASYNC-$applicationId"
                        addAppender(this@apply)
                        start()
                    }
                    logger.addAppender(asyncAppender)
                } else {
                    start()
                    logger.addAppender(this)
                }
            }
        }
    }
}
```

### 日志查询和分析

系统支持多维度的日志查询和分析：

```kotlin
class OperationLogServiceImpl(
    private val operationLogRepository: OperationLogRepository
) : OperationLogService {
    
    override suspend fun listLogs(
        applicationId: UUID?,
        userId: UUID?,
        module: String?,
        operation: String?,
        startTime: Instant?,
        endTime: Instant?,
        page: Int,
        size: Int
    ): Page<OperationLog> {
        val criteria = mutableListOf<Criterion>()
        
        applicationId?.let { criteria.add(OperationLog::applicationId.eq(it)) }
        userId?.let { criteria.add(OperationLog::userId.eq(it)) }
        module?.let { criteria.add(OperationLog::module.eq(it)) }
        operation?.let { criteria.add(OperationLog::operation.eq(it)) }
        
        if (startTime != null && endTime != null) {
            criteria.add(OperationLog::createdAt.between(startTime, endTime))
        } else {
            startTime?.let { criteria.add(OperationLog::createdAt.ge(it)) }
            endTime?.let { criteria.add(OperationLog::createdAt.le(it)) }
        }
        
        return operationLogRepository.findAll(
            criteria = criteria,
            sort = Sort.by(Sort.Direction.DESC, "createdAt"),
            page = page,
            size = size
        )
    }
    
    override suspend fun exportLogs(
        applicationId: UUID?,
        userId: UUID?,
        module: String?,
        operation: String?,
        startTime: Instant?,
        endTime: Instant?
    ): InputStream {
        val logs = listLogs(
            applicationId = applicationId,
            userId = userId,
            module = module,
            operation = operation,
            startTime = startTime,
            endTime = endTime,
            page = 1,
            size = Int.MAX_VALUE
        ).content
        
        val workbook = XSSFWorkbook()
        val sheet = workbook.createSheet("操作日志")
        
        // 创建表头
        val headerRow = sheet.createRow(0)
        val headers = listOf("ID", "用户", "应用", "模块", "操作", "方法", "路径", "参数", "结果", "状态", "错误信息", "IP", "用户代理", "耗时(ms)", "创建时间")
        headers.forEachIndexed { index, header ->
            headerRow.createCell(index).setCellValue(header)
        }
        
        // 填充数据
        logs.forEachIndexed { index, log ->
            val row = sheet.createRow(index + 1)
            row.createCell(0).setCellValue(log.id.toString())
            row.createCell(1).setCellValue(log.username ?: "")
            row.createCell(2).setCellValue(log.applicationId?.toString() ?: "")
            row.createCell(3).setCellValue(log.module ?: "")
            row.createCell(4).setCellValue(log.operation ?: "")
            row.createCell(5).setCellValue(log.method ?: "")
            row.createCell(6).setCellValue(log.path ?: "")
            row.createCell(7).setCellValue(log.params ?: "")
            row.createCell(8).setCellValue(log.result ?: "")
            row.createCell(9).setCellValue(log.status?.toString() ?: "")
            row.createCell(10).setCellValue(log.errorMsg ?: "")
            row.createCell(11).setCellValue(log.ip ?: "")
            row.createCell(12).setCellValue(log.userAgent ?: "")
            row.createCell(13).setCellValue(log.duration?.toString() ?: "")
            row.createCell(14).setCellValue(log.createdAt.toString())
        }
        
        val outputStream = ByteArrayOutputStream()
        workbook.write(outputStream)
        workbook.close()
        
        return ByteArrayInputStream(outputStream.toByteArray())
    }
    
    override suspend fun clearLogs(applicationId: UUID?, beforeTime: Instant): Int {
        val criteria = mutableListOf<Criterion>()
        
        applicationId?.let { criteria.add(OperationLog::applicationId.eq(it)) }
        criteria.add(OperationLog::createdAt.lt(beforeTime))
        
        return operationLogRepository.deleteAll(criteria)
    }
}
```

## 安全考虑

1. **日志敏感信息保护**：对敏感信息（如密码、令牌）进行脱敏处理
2. **日志访问控制**：严格控制日志查询和导出权限
3. **日志完整性**：确保日志记录的完整性，防止篡改
4. **日志存储安全**：日志文件存储在安全位置，防止未授权访问
5. **日志备份**：定期备份重要日志，防止数据丢失
